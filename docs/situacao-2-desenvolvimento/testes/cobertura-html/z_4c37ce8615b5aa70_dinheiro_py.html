1"""Valores monetários: sempre Decimal com duas casas (RN03), formato brasileiro R$ 1.234,56 (RNF02).""" 

2import re 

3from decimal import ROUND_HALF_UP, Decimal, InvalidOperation 

4 

5CENTAVO = Decimal("0.01") 

6ZERO = Decimal("0.00") 

7 

8_SOMENTE_MILHAR = re.compile(r"-?\d{1,3}(\.\d{3})+") 

9_NUMERO = re.compile(r"-?\d+(\.\d{0,2})?") # "4," (vírgula sem centavos) vale 4,00

12def quantizar(valor) -> Decimal: 

13 """Converte para Decimal com duas casas. Floats passam por str() para não herdar imprecisão.""" 

14 if valor is None: 

15 return ZERO 

16 if isinstance(valor, float): 

17 valor = str(valor) 

18 return Decimal(valor).quantize(CENTAVO, rounding=ROUND_HALF_UP)

21def somar(valores) -> Decimal: 

22 return sum((quantizar(v) for v in valores), ZERO)

25def interpretar_moeda(texto: str) -> Decimal: 

26 """Lê um valor digitado pelo usuário: "1.234,56", "1234,56", "1234.56" ou "R$ 1.234,56". 

27 

28 Levanta ValueError se o texto não for um valor válido com até duas casas decimais. 

29 """ 

30 limpo = (texto or "").replace("R$", "").replace(" ", "").replace(" ", "").strip() 

31 if not limpo: 

32 raise ValueError("Informe um valor.") 

33 if "," in limpo: 

34 limpo = limpo.replace(".", "").replace(",", ".") 

35 elif _SOMENTE_MILHAR.fullmatch(limpo): 

36 limpo = limpo.replace(".", "") 

37 if not _NUMERO.fullmatch(limpo): 

38 raise ValueError("Informe um valor válido, por exemplo 1.234,56.") 

39 try: 

40 return quantizar(limpo) 

41 except InvalidOperation as erro: # pragma: no cover - regex já garante formato 

42 raise ValueError("Informe um valor válido, por exemplo 1.234,56.") from erro

45def formatar_numero(valor) -> str: 

46 """1234.5 -> "1.234,50" (sem símbolo, usado para preencher campos de formulário).""" 

47 valor = quantizar(valor) 

48 sinal = "-" if valor < 0 else "" 

49 inteiro, centavos = f"{abs(valor):.2f}".split(".") 

50 return f"{sinal}{int(inteiro):,}".replace(",", ".") + f",{centavos}"

53def formatar_moeda(valor) -> str: 

54 """1234.5 -> "R$ 1.234,50"; negativos ficam "-R$ 1.234,50".""" 

55 texto = formatar_numero(valor) 

56 if texto.startswith("-"): 

57 return f"-R$ {texto[1:]}" 

58 return f"R$ {texto}"
