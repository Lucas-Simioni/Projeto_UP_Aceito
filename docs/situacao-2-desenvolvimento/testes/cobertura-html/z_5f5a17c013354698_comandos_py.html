47 @click.option("--convidados", default=150, show_default=True, help="Quantidade de convidados.") 

48 def popular_exemplo(convidados: int): 

49 """Cria a conta de exemplo da persona (Juliana & Rafael) com dados de demonstração.""" 

50 if contas.email_cadastrado(EMAIL_EXEMPLO): 

51 click.echo(f"A conta {EMAIL_EXEMPLO} já existe. Nada foi alterado.") 

52 return 

53 hoje = datas.hoje() 

54 usuario = contas.criar_conta("Juliana", EMAIL_EXEMPLO, SENHA_EXEMPLO) 

55 usuario.criado_em = usuario.criado_em - timedelta(days=120) 

56 casamento, _ = casamentos.salvar_casamento( 

57 usuario, nome_casal="Juliana & Rafael", data_casamento=somar_meses(hoje, 10), 

58 local="Espaço Jardim das Flores", orcamento_total=Decimal("60000"), convidados_estimados=150, 

59 ) 

60 categorias = {c.nome: c for c in casamento.categorias} 

61 for nome, valor in PLANEJADO.items(): 

62 categorias[nome].valor_planejado = Decimal(valor) 

63 

64 fornecedores = {} 

65 for categoria, nome, valor, status in FORNECEDORES: 

66 fornecedor = Fornecedor(categoria=categorias[categoria], nome=nome, valor_orcado=Decimal(valor), 

67 status=status, telefone="(11) 99999-0000") 

68 db.session.add(fornecedor) 

69 fornecedores[nome] = fornecedor 

70 db.session.commit() 

71 

72 for nome_fornecedor, descricao, valor, parcelas, meses in DESPESAS: 

73 fornecedor = fornecedores[nome_fornecedor] 

74 contratacao = somar_meses(hoje, -meses) 

75 despesa, _ = despesas.registrar_despesa(casamento, despesas.DadosDespesa( 

76 categoria_id=fornecedor.categoria_id, fornecedor_id=fornecedor.id, descricao=descricao, 

77 valor_total=Decimal(valor), data_contratacao=contratacao, num_parcelas=parcelas, 

78 primeiro_vencimento=contratacao + timedelta(days=10), 

79 )) 

80 # Paga as parcelas vencidas, menos uma, para o painel mostrar um atraso real. 

81 vencidas = [p for p in despesa.parcelas if p.data_vencimento < hoje] 

82 for parcela in vencidas[:-1] if meses == 2 else vencidas: 

83 despesas.pagar_parcela(parcela, parcela.data_vencimento, hoje)
